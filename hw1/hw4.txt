    "            reward = 1\n",
    "            if self.state == \"D\":\n",
    "                nextState = \"T\" # terminal state\n",

    "def firstVisitMonteCarlo(env, num_episodes=1000):\n",
    "    # Initialize state-action value function\n",
    "    V = {state: 0.0 for state in env.states}\n",
    "    returns = {state: [] for state in env.states}\n",
    "    \n",
    "    for ep in range(1, num_episodes+1):\n",
    "        # Run episode\n",
    "        episodeData = []\n",
    "        state = env.start()\n",
    "        episodeDone = False\n",
    "        while not episodeDone:\n",
    "            if random.random() < 0.8:\n",
    "                action = \"R\"\n",
    "            else:\n",
    "                action = \"L\"\n",
    "            next_state, reward, episodeDone = env.step(action)\n",
    "            episodeData.append((state, action, reward))\n",
    "            state = next_state\n",
    "        \n",
    "        trajectory = \" → \".join(f\"{s}_{a,r}\" for s,a,r in episodeData)\n",
    "        trajectory += \" → Terminal\"\n",
    "        print(f\"Episode {ep}: {trajectory}\")\n",
    "            \n",
    "        G = 0\n",
    "        visit = set()\n",
    "        # go through the episode in reverse\n",
    "        for s,_,r in reversed(episodeData):\n",
    "            G = r + G\n",
    "            if s not in visit:\n",
    "                visit.add(s)\n",
    "                returns[s].append(G)\n",
    "                V[s] = sum(returns[s]) / len(returns[s])\n",
    "    return V\n"
